/* ================= 答题 ================= */
function buildTicks(){
  var box = $('ticks');
  if(box.childElementCount === 18) return;
  box.innerHTML = '';
  for(var n=0;n<18;n++){
    var d = document.createElement('div');
    d.className = 'tick';
    box.appendChild(d);
  }
}
function paintTicks(){
  var ticks = $('ticks').children;
  for(var n=0;n<18;n++){
    var t = ticks[n];
    t.className = 'tick' + (n===state.i ? ' now' : (state.answers[n]!==null ? ' done' : ''));
  }
}
function renderQuestion(){
  var q = QUESTIONS[state.i];
  $('qNum').textContent = '第 ' + pad(state.i+1) + ' 题';
  $('qNow').textContent = pad(state.i+1);
  $('qDim').textContent = state.i === 17 ? '最后一题' : '剩余 ' + (17 - state.i) + ' 题';
  $('qText').textContent = q.q;
  var box = $('opts');
  box.innerHTML = '';
  q.opts.forEach(function(o, idx){
    var btn = document.createElement('button');
    btn.type = 'button';
    btn.className = 'opt' + (state.pick[state.i] === idx ? ' sel' : '');
    btn.setAttribute('aria-pressed', 'false');
    btn.innerHTML = '<span class="k">' + 'ABCD'[idx] + '</span><span class="t"></span>';
    btn.querySelector('.t').textContent = o[0];
    btn.addEventListener('click', function(){ choose(idx); });
    box.appendChild(btn);
  });
  $('backBtn').disabled = (state.i === 0);
  state.locked = false;
  paintTicks();
}
function choose(idx){
  if(state.locked) return;
  var q = QUESTIONS[state.i];
  state.locked = true;
  state.answers[state.i] = q.opts[idx][1];
  state.pick[state.i] = idx;
  var all = $('opts').children;
  Array.prototype.forEach.call(all, function(el){
    var on = el === all[idx];
    el.classList.toggle('sel', on);
    el.setAttribute('aria-pressed', on ? 'true' : 'false');
  });
  paintTicks();
  setTimeout(function(){
    if(state.i === 17){ finish(); }
    else { state.i++; renderQuestion(); }
  }, 230);
}
$('backBtn').addEventListener('click', function(){
  if(state.i === 0) return;
  state.i--;
  renderQuestion();
});
window.addEventListener('keydown', function(e){
  if(!byId.quiz.classList.contains('is-active')) return;
  var idx = ['1','2','3','4'].indexOf(e.key);
  if(idx < 0) idx = ['a','b','c','d'].indexOf(e.key.toLowerCase());
  if(idx >= 0){ choose(idx); }
  if(e.key === 'ArrowLeft'){ $('backBtn').click(); }
});

/* ================= 计算 ================= */
function finish(){
  var total = 0;
  state.answers.forEach(function(v){ total += v; });
  // 作答倾向（14–50 岁区间）与实际年龄轻度合成，使结果落在合理区间
  var raw = 14 + (total - 18) * 0.68;
  state.psych = Math.round(raw * 0.7 + state.age * 0.3);
  ORDER.forEach(function(k){
    var s = 0;
    QUESTIONS.forEach(function(q, i){ if(q.dim === k) s += state.answers[i]; });
    state.dimPct[k] = Math.round((s - 3) / 9 * 100);
    state.dimPct[k] = Math.max(0, Math.min(100, state.dimPct[k]));
  });
  showView('result');
  renderResult();
}

function bandLabel(p){ return p >= 72 ? '偏强' : (p >= 44 ? '平衡' : '偏弱'); }

/* 年龄对照轴 */
function axisHtml(age, pa){
  var lo = Math.floor((Math.min(age, pa) - 7) / 5) * 5;
  var hi = Math.ceil((Math.max(age, pa) + 7) / 5) * 5;
  if(hi - lo < 20){ hi = lo + 20; }
  var pAge = (age - lo) / (hi - lo) * 100;
  var pPa = (pa - lo) / (hi - lo) * 100;
  var left = Math.min(pAge, pPa), width = Math.abs(pAge - pPa);
  return '<div class="axis">' +
    '<div class="axis-track">' +
      '<div class="axis-band" style="left:' + left.toFixed(1) + '%;width:' + width.toFixed(1) + '%"></div>' +
      '<div class="mark actual" style="left:' + pAge.toFixed(1) + '%"><span class="tag">实际 ' + age + '</span><span class="dot"></span></div>' +
      '<div class="mark psych" style="left:' + pPa.toFixed(1) + '%"><span class="dot"></span><span class="tag">心理 ' + pa + '</span></div>' +
      '<span class="axis-end l">' + lo + ' 岁</span>' +
      '<span class="axis-end r">' + hi + ' 岁</span>' +
    '</div>' +
  '</div>';
}

/* ================= 雷达图 ================= */
function drawRadar(dims){
  var el = $('radar');
  if(!el) return;
  if(typeof echarts === 'undefined'){
    var tries = 0;
    (function wait(){
      if(typeof echarts !== 'undefined'){ paint(); return; }
      if(++tries > 40){ el.style.display = 'none'; return; }
      setTimeout(wait, 150);
    })();
    return;
  }
  paint();

  function paint(){
    var chart = echarts.init(el, null, { renderer:'canvas' });
    chart.setOption({
      animationDuration: 1000,
      animationEasing: 'cubicOut',
      silent: true,
      radar: {
        center: ['50%','52%'],
        radius: '61%',
        splitNumber: 4,
        shape: 'polygon',
        indicator: dims.map(function(x){ return { name:x.short, max:100 }; }),
        axisName: {
          color:'#33433D',
          fontSize:11.5,
          fontFamily:'"Noto Sans SC","PingFang SC",sans-serif',
          lineHeight:16
        },
        nameGap: 10,
        splitLine: { lineStyle:{ color:'#E3E6E0', width:1 } },
        splitArea: { areaStyle:{ color:['rgba(44,106,88,0.05)','rgba(250,250,248,0)'] } },
        axisLine: { lineStyle:{ color:'#E3E6E0' } }
      },
      series: [{
        type:'radar',
        symbol:'circle',
        symbolSize:4.5,
        lineStyle:{ color:'#2C6A58', width:1.4 },
        itemStyle:{ color:'#2C6A58', borderColor:'#2C6A58' },
        areaStyle:{ color:'rgba(44,106,88,0.10)' },
        data:[{ value: dims.map(function(x){ return x.pct; }) }]
      }]
    });
    window.addEventListener('resize', function(){ chart.resize(); });
  }
}

})();
